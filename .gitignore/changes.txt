/* Parse "09.15" → {h:9, m:15} */
function parseDotTime(s) {
  const p = String(s || '').split('.');
  return { h: parseInt(p[0], 10) || 0, m: parseInt(p[1] || '0', 10) || 0 };
}

/* Build a Date for (baseY, baseM0, baseD) + offsetDays at HH.MM */
function slotDate(baseY, baseM0, baseD, offsetDays, dotTime) {
  const t = parseDotTime(dotTime);
  return new Date(baseY, baseM0, baseD + offsetDays, t.h, t.m, 0, 0);
}

/* Get or create the primary LifeOS calendar */

/* Title prefix by item kind */
function slotEmoji(kind) {
  if (kind === 'birthday')    return '🎂';
  if (kind === 'anniversary') return '💍';
  if (kind === 'countdown')   return '🔥';
  return '🔔';
}

/* Compose event title for a slot */
function slotTitle(kind, title, slot) {
  const em = slotEmoji(kind);
  return em + ' ' + slot.label + ': ' + title;
}

/* Delete events matching this title in the window [from, to] on the given calendar. */
function deleteEventsByTitle(cal, title, from, to) {
  let n = 0;
  try {
    const evs = cal.getEvents(from, to);
    evs.forEach(function (ev) {
      if (ev.getTitle() === title) {
        try { ev.deleteEvent(); n++; } catch (_) {}
      }
    });
  } catch (_) {}
  return n;
}

/* Core: ensure the 5 slots exist for one item.
   item = { kind, title, dateY, dateM0, dateD, notes, sourceId } */
function ensureFiveSlots(item) {
  const cal = getOrCreateLifeOSCalendar();
  const y = item.dateY, m0 = item.dateM0, d = item.dateD;
  const created = [];
  const skipped = [];

  SLOT_DEFS.forEach(function (slot) {
    const start = slotDate(y, m0, d, slot.offsetDays, slot.start);
    const end   = slotDate(y, m0, d, slot.offsetDays, slot.end);
    const title = slotTitle(item.kind, item.title, slot);

    // Skip if it's in the past
    if (end.getTime() < Date.now()) { skipped.push(slot.n); return; }

    // Dedupe window: ±1 hour around start
    const from = new Date(start.getTime() - 3600000);
    const to   = new Date(end.getTime()   + 3600000);
    let exists = false;
    try {
      const evs = cal.getEvents(from, to);
      for (let i = 0; i < evs.length; i++) {
        if (evs[i].getTitle() === title) {
          const s = evs[i].getStartTime().getTime();
          if (Math.abs(s - start.getTime()) < 60000) { exists = true; break; }
        }
      }
    } catch (_) {}

    if (exists) { skipped.push(slot.n); return; }

    try {
      const ev = cal.createEvent(title, start, end, {
        description: item.notes || ('Auto reminder · ' + item.kind)
      });
      try { ev.removeAllReminders(); } catch (_) {}
      try { ev.addPopupReminder(2); } catch (_) {}
      created.push(slot.n);
    } catch (err) {
      Logger.log('ensureFiveSlots create fail ' + title + ' → ' + err);
    }
  });

  return { created: created, skipped: skipped };
}

/* Delete all 5 slots for one item */
function deleteFiveSlots(kind, title, y, m0, d) {
  const cal = getOrCreateLifeOSCalendar();
  let removed = 0;
  SLOT_DEFS.forEach(function (slot) {
    const start = slotDate(y, m0, d, slot.offsetDays, slot.start);
    const end   = slotDate(y, m0, d, slot.offsetDays, slot.end);
    const titleStr = slotTitle(kind, title, slot);
    const from = new Date(start.getTime() - 3600000);
    const to   = new Date(end.getTime()   + 3600000);
    removed += deleteEventsByTitle(cal, titleStr, from, to);
  });
  return removed;
}

    status: 'success',
    eventsCreated: totalCreated,

    eventsDeleted: deletedEvents
