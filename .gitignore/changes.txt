/* Normalize a task name → array of comparable tokens */
function _normalizeTaskName(s) {
  if (!s) return [];
  let t = String(s).toLowerCase();
  // strip parentheticals
  t = t.replace(/\([^)]*\)/g, ' ');
  // strip non-alphanumeric
  t = t.replace(/[^a-z0-9\s]/g, ' ');
  // split
  let tokens = t.split(/\s+/).filter(function (x) { return x.length > 0; });
  // drop noise tokens
  tokens = tokens.filter(function (x) { return RECONCILE_NOISE.indexOf(x) === -1; });
  // expand synonyms → keep canonical + synonyms for matching
  const out = [];
  tokens.forEach(function (tok) {
    if (RECONCILE_SYNONYMS[tok]) {
      RECONCILE_SYNONYMS[tok].forEach(function (syn) {
        if (out.indexOf(syn) === -1) out.push(syn);
      });
    } else {
      if (out.indexOf(tok) === -1) out.push(tok);
    }
  });
  return out;
}

/* Similarity score: |intersection| / |union|  (Jaccard on token sets) */
function _similarityScore(aTokens, bTokens) {
  if (!aTokens.length || !bTokens.length) return 0;
  const aSet = {};
  aTokens.forEach(function (t) { aSet[t] = true; });
  const bSet = {};
  bTokens.forEach(function (t) { bSet[t] = true; });

  let inter = 0;
  Object.keys(aSet).forEach(function (t) {
    if (bSet[t]) inter++;
  });
  const union = Object.keys(aSet).length + Object.keys(bSet).length - inter;
  if (union === 0) return 0;
  return inter / union;
}

/* Find best match for `oldName` among active MASTER DATA task names.
   Returns { name, score } or null. */
function _findBestMatch(oldName, activeTasks) {
  const oldTokens = _normalizeTaskName(oldName);
  if (!oldTokens.length) return null;

  let best = null;
  activeTasks.forEach(function (t) {
    const score = _similarityScore(oldTokens, _normalizeTaskName(t.area));
    if (!best || score > best.score) best = { task: t, score: score };
  });

  if (best && best.score >= 0.65) return { name: best.task.area, score: best.score };
  return null;
}

/* Read the whole tracker, split into 4 groups:
   - activeRows:   matching a current MASTER DATA task
   - inactiveRows: not matching (or the archive zone)
   - summaryInfo:  metadata about SUMMARY block (row indexes)
   - abandonInfo:  metadata about ABANDONED block
   Also returns the header row and column count. */
function _readTrackerStructure(sheet) {
  const data = sheet.getDataRange().getValues();
  if (!data.length) return null;

  const headerRow = data[0];
  const numCols = headerRow.length;

  const result = {
    headerRow: headerRow,
    numCols: numCols,
    activeRows: [],       // { head, subCat, area, type, altDays, data: [...cells from col 5 onwards] }
    inactiveRows: [],     // same shape, with the 🗄️ prefix stripped
    summaryBlock: null,   // { rowIdx, rows: [...] }  (raw rows to preserve formatting)
    abandonBlock: null
  };

  let inSummary = false;
  let inAbandon = false;

  for (let i = 1; i < data.length; i++) {
    const a = String(data[i][0] || '').trim();
    const b = String(data[i][1] || '').trim();

    // Summary block start
    if (a === '📊 SUMMARY') {
      inSummary = true;
      result.summaryBlock = { rows: [data[i].slice()] };
      continue;
    }
    // Abandon block start
    if (a === '📊 ABANDONED REASON') {
      inSummary = false;
      inAbandon = true;
      result.abandonBlock = { rows: [data[i].slice()] };
      continue;
    }
    // Keep absorbing rows for whichever block we're in
    if (inSummary) {
      result.summaryBlock.rows.push(data[i].slice());
      // summary is ~5 rows total (header + 4 stat rows)
      if (result.summaryBlock.rows.length >= 5) inSummary = false;
      continue;
    }
    if (inAbandon) {
      result.abandonBlock.rows.push(data[i].slice());
      if (result.abandonBlock.rows.length >= 2) inAbandon = false;
      continue;
    }

    // Skip empty rows
    if (!a && !b && !data[i][2] && !data[i][3]) continue;

    // Regular task row
    const head = a;
    const area = String(data[i][2] || '').trim();
    if (!area) continue;

    // Detect archive prefix
    const isArchived = head.indexOf('🗄️') === 0;
    const cleanHead = isArchived ? head.replace(/^🗄️\s*/, '') : head;

    const row = {
      head: cleanHead,
      subCat: String(data[i][1] || '').trim(),
      area: area,
      type: String(data[i][3] || '').trim(),
      altDays: String(data[i][4] || ''),
      tail: data[i].slice(5),       // date columns onwards
      isArchived: isArchived
    };

    if (isArchived) {
      result.inactiveRows.push(row);
    } else {
      result.activeRows.push(row);
    }
  }
  return result;
}
