/* [106b] Install the auto-clean trigger. Run ONCE from Apps Script
   editor. It calls cleanAllPastEvents() every 15 minutes so past
   events disappear automatically. */
function installPastEventCleaner() {
  // Remove any existing triggers for the same function first
  const triggers = ScriptApp.getProjectTriggers();
  triggers.forEach(function (t) {
    if (t.getHandlerFunction() === 'cleanAllPastEvents') {
      ScriptApp.deleteTrigger(t);
    }
  });

  ScriptApp.newTrigger('cleanAllPastEvents')
    .timeBased()
    .everyMinutes(15)
    .create();

  Logger.log('✅ Installed 15-min trigger for cleanAllPastEvents');
}

/* [106c] Install a DAILY trigger for reminderSyncAll — run ONCE from
   the Apps Script editor. Ensures birthday/anniversary slots roll to
   next year automatically, and any stale events get cleaned. */
function installDailyReminderSync() {
  const triggers = ScriptApp.getProjectTriggers();
  triggers.forEach(function (t) {
    if (t.getHandlerFunction() === 'reminderSyncAll') {
      ScriptApp.deleteTrigger(t);
    }
  });

  ScriptApp.newTrigger('reminderSyncAll')
    .timeBased()
    .everyDays(1)
    .atHour(2)
    .create();

  Logger.log('✅ Installed daily 02:00 trigger for reminderSyncAll');
}
