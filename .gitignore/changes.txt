/* [43] */ function getNotesHistory(days) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('EVERYDAY_NOTES');
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  const out = [];
  for (let i = data.length - 1; i >= 1 && out.length < days; i--) {
    let d = data[i][0];
    if (d instanceof Date) d = Utilities.formatDate(d, Session.getScriptTimeZone(), 'dd-MMM-yyyy');
    if (!d) continue;
    out.push({
      date: String(d),
      positive: textToItems(data[i][1]),
      improve:  textToItems(data[i][2]),
      common:   textToItems(data[i][3]),
      learned:  textToItems(data[i][4]),
      office:   textToItems(data[i][5])
    });
  }
  return out;
}
